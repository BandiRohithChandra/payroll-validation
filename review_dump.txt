

==================== backend/datagen/pipeline.py ====================

"""One function that builds the whole dataset in memory, in dependency order."""
import random

import config
from datagen.anomalies import inject_input_anomalies, inject_source_anomalies
from datagen.evaluate import evaluate
from datagen.ids import IdGenerator
from datagen.leave import generate_leave_records
from datagen.master import generate_employees, generate_pay_groups, generate_pay_periods
from datagen.pay_items import generate_allowances, generate_deductions
from datagen.payroll import aggregate_payroll_inputs
from datagen.rules import RULES, run_validation
from datagen.time_entries import generate_time_entries
from datagen.workflow import run_workflow, summarize


def build_dataset(scale_name="starter", seed=config.SEED):
    scale = config.SCALES[scale_name]
    rng, ids = random.Random(seed), IdGenerator()

    employees, p_emp = generate_employees(rng, ids, scale)
    time_entries, p_time = generate_time_entries(rng, employees)
    leaves = generate_leave_records(rng, ids, employees)
    allowances = generate_allowances(rng, ids, employees, time_entries)
    deductions = generate_deductions(rng, ids, employees, leaves)

    src_truth, forced, used = inject_source_anomalies(
        rng, ids, scale, employees, time_entries, allowances, deductions)
    payroll_inputs = aggregate_payroll_inputs(employees, time_entries, allowances,
                                              deductions, forced)
    in_truth = inject_input_anomalies(rng, scale, payroll_inputs, used)

    truth = p_emp + p_time + src_truth + in_truth
    findings = run_validation(employees, time_entries, allowances, deductions, payroll_inputs)

    pay_groups, pay_periods = generate_pay_groups(), generate_pay_periods()
    flow = run_workflow(rng, ids, employees, time_entries, allowances, deductions,
                        payroll_inputs, pay_periods, findings)
    return {
        "pay_groups": pay_groups, "pay_periods": pay_periods,
        "employees": employees, "time_entries": time_entries, "leaves": leaves,
        "allowances": allowances, "deductions": deductions,
        "payroll_inputs": payroll_inputs, "validation_rules": RULES,
        "findings": findings, "truth": truth, **flow,
    }


if __name__ == "__main__":
    import sys
    data = build_dataset(sys.argv[1] if len(sys.argv) > 1 else "starter")
    print({k: len(v) for k, v in data.items()})
    for row in evaluate(data["findings"], data["truth"]):
        print(row)
    print(summarize(data["payroll_inputs"], data["exceptions"], data["payroll_exports"]))
    exported = [r for r in data["payroll_inputs"] if r["status"] == "exported"]
    print("exported with unapproved source rows:",
          sum(r["unapproved_source_count"] > 0 for r in exported))


==================== backend/datagen/payroll.py ====================

"""PayrollInput aggregation: one row per active employee per pay period.
Only approved source rows count (see docs/decisions.md)."""
from collections import defaultdict
from datetime import datetime, time, timedelta
from decimal import ROUND_HALF_UP, Decimal

import config
from datagen.ids import IdGenerator
from datagen.pay_items import _active
from datagen.time_entries import _working_days, effective_time_entries

CENT = Decimal("0.01")
ZERO = Decimal("0.00")


def money(x):
    return x.quantize(CENT, rounding=ROUND_HALF_UP)


def counted_time_entries(time_entries, force_unapproved=frozenset()):
    """(employee_id, work_date) -> entry that counts. Normally the approved
    effective entry. For (employee_id, pay_period_id) keys in force_unapproved,
    non-approved originals are also counted: this simulates the injected
    'unapproved time exported' defect."""
    counted = dict(effective_time_entries(time_entries))
    for e in time_entries:
        if (e["entry_type"] == "original" and e["approval_status"] != "approved"
                and (e["employee_id"], e["pay_period_id"]) in force_unapproved):
            counted.setdefault((e["employee_id"], e["work_date"]), e)
    return counted


def aggregate_payroll_inputs(employees, time_entries, allowances, deductions,
                             force_unapproved=frozenset(), exclude_ids=frozenset()):
    counted = counted_time_entries(time_entries, force_unapproved)
    entries = defaultdict(list)
    for e in counted.values():
        entries[(e["employee_id"], e["pay_period_id"])].append(e)

    allow, ded = defaultdict(lambda: ZERO), defaultdict(lambda: ZERO)
    for a in allowances:
        if a["approval_status"] == "approved" and a["allowance_id"] not in exclude_ids:
            allow[(a["employee_id"], a["pay_period_id"])] += a["amount"]
    for d in deductions:
        if d["approval_status"] == "approved" and d["deduction_id"] not in exclude_ids:
            ded[(d["employee_id"], d["pay_period_id"])] += d["amount"]

    rows = []
    for period in config.PAY_PERIODS:
        working = _working_days(period)
        month = period.start_date.month
        for emp in sorted(employees, key=lambda e: e["employee_id"]):
            if not _active(emp, period):
                continue
            eid = emp["employee_id"]
            key = (eid, period.pay_period_id)
            group = config.PAY_GROUP_BY_ID[emp["pay_group_id"]]
            rows_in = entries.get(key, [])
            unapproved = sum(1 for e in rows_in if e["approval_status"] != "approved")

            reg = ot = ZERO
            if emp["pay_type"] == "hourly":
                for e in rows_in:
                    h = e["hours_worked"]
                    r = min(h, group.standard_hours_per_day)
                    reg += r
                    ot += h - r
                base = money(reg * emp["hourly_rate"])
                overtime = money(ot * emp["hourly_rate"] * group.overtime_multiplier)
            else:
                last = emp["termination_date"] or period.end_date
                days = sum(1 for d in working if emp["hire_date"] <= d <= last)
                base = money(emp["monthly_salary"] * days / period.working_days)
                overtime = ZERO

            allowance_total = money(allow.get(key, ZERO))
            deduction_total = money(ded.get(key, ZERO))
            gross = base + overtime + allowance_total
            rows.append({
                "payroll_input_id": IdGenerator.payroll_input_id(eid, month),
                "employee_id": eid,
                "pay_period_id": period.pay_period_id,
                "pay_group_id": emp["pay_group_id"],
                "pay_type": emp["pay_type"],
                "regular_hours": reg,
                "overtime_hours": ot,
                "base_pay": base,
                "overtime_pay": overtime,
                "allowance_total": allowance_total,
                "deduction_total": deduction_total,
                "gross_pay": gross,
                "net_pay": gross - deduction_total,
                "unapproved_source_count": unapproved,
                "status": "draft",
                "aggregated_at": datetime.combine(period.end_date + timedelta(days=8),
                                                  time(9, 0)),
            })
    return rows


==================== backend/datagen/rules.py ====================

"""ValidationRule table and the validation engine."""
from collections import defaultdict

import config
from datagen.payroll import aggregate_payroll_inputs


def _rule(rule_id, name, severity, table, description):
    return {"rule_id": rule_id, "name": name, "severity": severity, "target_table": table,
            "description": description, "status": "active",
            "effective_from": config.PERIOD_START}


RULES = [
    _rule("VR-001", "Missing bank details", "blocking", "employees",
          "Bank account number or IFSC is empty."),
    _rule("VR-002", "Missing tax id", "blocking", "employees", "Tax id is empty."),
    _rule("VR-003", "Duplicate time submission", "warning", "time_entries",
          "More than one non-correction entry for the same employee and date; the later one is flagged."),
    _rule("VR-004", "Hours exceed period maximum", "blocking", "payroll_inputs",
          "Regular plus overtime hours are above the pay group maximum."),
    _rule("VR-005", "Daily hours outlier", "warning", "time_entries",
          "An original entry has more hours than the daily outlier threshold."),
    _rule("VR-006", "Duplicate allowance", "blocking", "allowances",
          "More than one approved allowance of the same type for an employee and period."),
    _rule("VR-007", "Unapproved source in payroll input", "blocking", "payroll_inputs",
          "The input counts source rows that are not approved."),
    _rule("VR-008", "Leave deduction applied twice", "blocking", "deductions",
          "More than one approved unpaid_leave deduction for the same leave record."),
    _rule("VR-009", "Payroll totals do not reconcile", "blocking", "payroll_inputs",
          "Stored totals differ from a recomputation from approved source rows, or net is not gross minus deductions."),
    _rule("VR-010", "Late time submission", "warning", "time_entries",
          "Entry submitted after the pay period submission deadline."),
]
SEVERITY = {r["rule_id"]: r["severity"] for r in RULES}
RECON_FIELDS = ("regular_hours", "overtime_hours", "base_pay", "overtime_pay",
                "allowance_total", "deduction_total", "gross_pay", "net_pay")


def run_validation(employees, time_entries, allowances, deductions, payroll_inputs):
    findings = []
    tol = config.RECONCILIATION_TOLERANCE
    period_by_id = {p.pay_period_id: p for p in config.PAY_PERIODS}

    def flag(rule, table, record_id, employee_id, period_id, detail):
        findings.append({"rule_id": rule, "severity": SEVERITY[rule], "table_name": table,
                         "record_id": record_id, "employee_id": employee_id,
                         "pay_period_id": period_id, "detail": detail})

    for e in employees:
        eid = e["employee_id"]
        if not e["bank_account_number"] or not e["bank_ifsc"]:
            flag("VR-001", "employees", eid, eid, None, "bank account or IFSC missing")
        if not e["tax_id"]:
            flag("VR-002", "employees", eid, eid, None, "tax id missing")

    same_day = defaultdict(list)
    for t in time_entries:
        if t["entry_type"] != "correction":
            same_day[(t["employee_id"], t["work_date"])].append(t)
        if t["entry_type"] == "original":
            if t["hours_worked"] > config.DAILY_HOURS_OUTLIER:
                flag("VR-005", "time_entries", t["time_entry_id"], t["employee_id"],
                     t["pay_period_id"], f"{t['hours_worked']} hours in one day")
            if t["submitted_at"].date() > period_by_id[t["pay_period_id"]].submission_deadline:
                flag("VR-010", "time_entries", t["time_entry_id"], t["employee_id"],
                     t["pay_period_id"], "submitted after the deadline")
    for rows in same_day.values():
        if len(rows) > 1:
            rows.sort(key=lambda r: (r["submitted_at"], r["time_entry_id"]))
            for r in rows[1:]:
                flag("VR-003", "time_entries", r["time_entry_id"], r["employee_id"],
                     r["pay_period_id"], f"same day as {rows[0]['time_entry_id']}")

    allowance_groups = defaultdict(list)
    for a in allowances:
        if a["approval_status"] == "approved":
            allowance_groups[(a["employee_id"], a["pay_period_id"], a["allowance_type"])].append(a)
    for rows in allowance_groups.values():
        rows.sort(key=lambda r: r["allowance_id"])
        for r in rows[1:]:
            flag("VR-006", "allowances", r["allowance_id"], r["employee_id"],
                 r["pay_period_id"], f"{r['allowance_type']} already counted in {rows[0]['allowance_id']}")

    leave_groups = defaultdict(list)
    for d in deductions:
        if d["deduction_type"] == "unpaid_leave" and d["approval_status"] == "approved":
            leave_groups[d["leave_id"]].append(d)
    for rows in leave_groups.values():
        rows.sort(key=lambda r: r["deduction_id"])
        for r in rows[1:]:
            flag("VR-008", "deductions", r["deduction_id"], r["employee_id"],
                 r["pay_period_id"], f"leave {r['leave_id']} already deducted in {rows[0]['deduction_id']}")

    expected = {r["payroll_input_id"]: r
                for r in aggregate_payroll_inputs(employees, time_entries, allowances, deductions)}
    for r in payroll_inputs:
        pid, eid, per = r["payroll_input_id"], r["employee_id"], r["pay_period_id"]
        hours = r["regular_hours"] + r["overtime_hours"]
        limit = config.PAY_GROUP_BY_ID[r["pay_group_id"]].max_hours_per_period
        if hours > limit:
            flag("VR-004", "payroll_inputs", pid, eid, per, f"{hours} hours, maximum {limit}")
        if r["unapproved_source_count"] > 0:
            flag("VR-007", "payroll_inputs", pid, eid, per,
                 f"{r['unapproved_source_count']} unapproved source rows counted")
        exp = expected.get(pid)
        bad = exp is None or any(abs(r[k] - exp[k]) > tol for k in RECON_FIELDS)
        bad = bad or abs(r["gross_pay"] - r["deduction_total"] - r["net_pay"]) > tol
        if bad:
            flag("VR-009", "payroll_inputs", pid, eid, per, "stored totals differ from recomputation")
    return findings


==================== backend/datagen/anomalies.py ====================

"""Labelled edge cases injected into the valid baseline (see docs/decisions.md).
Source-level cases are injected before aggregation, input-level cases after.
Each case gets its own employee-period, so one label explains one defect."""
from collections import defaultdict
from datetime import timedelta
from decimal import Decimal

import config
from datagen.ids import IdGenerator
from datagen.time_entries import effective_time_entries

D = Decimal
HOURS_FILL = D("12.00")


def _truth(issue, table, record_id, rule, note):
    return {"issue_type": issue, "table_name": table, "record_id": record_id,
            "expected_rule": rule, "injected": True, "note": note}


def _pin(employee_id, period_id):
    return IdGenerator.payroll_input_id(employee_id, int(period_id[-2:]))


def _pick(rng, candidates, n, used, key_of):
    """Random choice of up to n candidates whose (employee, period) key is unused."""
    chosen = []
    for c in rng.sample(candidates, len(candidates)):
        k = key_of(c)
        if k in used:
            continue
        used.add(k)
        chosen.append(c)
        if len(chosen) == n:
            break
    return chosen


def _duplicate(rng, row, id_field, related_field, new_id):
    dup = dict(row)
    dup[id_field] = new_id
    dup["entry_type"] = "duplicate_submission"
    dup[related_field] = row[id_field]
    dup["created_at"] = row["created_at"] + timedelta(hours=rng.randint(1, 5))
    dup["approved_at"] = dup["created_at"] + timedelta(hours=rng.randint(1, 72))
    return dup


def inject_source_anomalies(rng, ids, scale, employees, time_entries, allowances, deductions):
    """Edits and appends to the lists in place.
    Returns (ground_truth, force_unapproved, used_keys)."""
    n = scale.anomalies_per_type
    truth, used = [], set()
    emp_by_id = {e["employee_id"]: e for e in employees}
    max_hours = lambda eid: config.PAY_GROUP_BY_ID[emp_by_id[eid]["pay_group_id"]].max_hours_per_period

    by_key = defaultdict(list)  # (employee, period) -> effective entries
    for e in effective_time_entries(time_entries).values():
        by_key[(e["employee_id"], e["pay_period_id"])].append(e)

    # 1. hours exceed the period maximum -> VR-004 on the payroll input
    over = sorted(k for k, rows in by_key.items()
                  if len(rows) * HOURS_FILL > max_hours(k[0])
                  and all(r["hours_worked"] <= HOURS_FILL for r in rows))
    for k in _pick(rng, over, n, used, lambda k: k):
        for r in by_key[k]:
            r["hours_worked"] = HOURS_FILL
        truth.append(_truth("hours_exceed_period_max", "payroll_inputs", _pin(*k), "VR-004",
                            "effective approved entries raised to 12h; total above group maximum"))

    # 2. unapproved time exported -> VR-007, and totals no longer reconcile -> VR-009
    pending = defaultdict(D)
    for e in time_entries:
        if e["entry_type"] == "original" and e["approval_status"] != "approved":
            pending[(e["employee_id"], e["pay_period_id"])] += e["hours_worked"]
    fits = sorted(k for k, h in pending.items()
                  if sum((r["hours_worked"] for r in by_key.get(k, [])), D("0")) + h <= max_hours(k[0]))
    forced = set()
    for k in _pick(rng, fits, n, used, lambda k: k):
        forced.add(k)
        for rule in ("VR-007", "VR-009"):
            truth.append(_truth("unapproved_time_exported", "payroll_inputs", _pin(*k), rule,
                                "non-approved time counted in the payroll input"))

    # 3. same allowance duplicated -> VR-006 on the duplicate row
    approved = sorted((a for a in allowances if a["approval_status"] == "approved"),
                      key=lambda a: a["allowance_id"])
    for a in _pick(rng, approved, n, used, lambda a: (a["employee_id"], a["pay_period_id"])):
        dup = _duplicate(rng, a, "allowance_id", "related_allowance_id", ids.next("allowance"))
        allowances.append(dup)
        truth.append(_truth("duplicate_allowance", "allowances", dup["allowance_id"], "VR-006",
                            f"copy of {a['allowance_id']}"))

    # 4. leave deduction applied twice -> VR-008 on the duplicate row
    unpaid = sorted((d for d in deductions
                     if d["deduction_type"] == "unpaid_leave" and d["approval_status"] == "approved"),
                    key=lambda d: d["deduction_id"])
    for d in _pick(rng, unpaid, n, used, lambda d: (d["employee_id"], d["pay_period_id"])):
        dup = _duplicate(rng, d, "deduction_id", "related_deduction_id", ids.next("deduction"))
        deductions.append(dup)
        truth.append(_truth("leave_deduction_twice", "deductions", dup["deduction_id"], "VR-008",
                            f"copy of {d['deduction_id']} for {d['leave_id']}"))
    return truth, frozenset(forced), used


def inject_input_anomalies(rng, scale, payroll_inputs, used):
    """5. payroll totals fail to reconcile -> VR-009. Edits net_pay in place."""
    truth = []
    candidates = sorted(r["payroll_input_id"] for r in payroll_inputs
                        if (r["employee_id"], r["pay_period_id"]) not in used)
    by_id = {r["payroll_input_id"]: r for r in payroll_inputs}
    for pid in sorted(rng.sample(candidates, min(scale.anomalies_per_type, len(candidates)))):
        delta = D(rng.choice([-2500, -1200, -350, 125, 480, 1750, 3000])) + D(rng.randrange(1, 100)) / 100
        by_id[pid]["net_pay"] += delta
        truth.append(_truth("totals_fail_to_reconcile", "payroll_inputs", pid, "VR-009",
                            f"net_pay shifted by {delta}"))
    return truth


==================== backend/datagen/evaluate.py ====================

"""Compares validation findings with ground truth, per rule."""
from collections import defaultdict


def evaluate(findings, truth):
    flagged, expected = defaultdict(set), defaultdict(set)
    for f in findings:
        flagged[f["rule_id"]].add(f["record_id"])
    for t in truth:
        expected[t["expected_rule"]].add(t["record_id"])
    rows = []
    for rule in sorted(set(flagged) | set(expected)):
        tp = len(flagged[rule] & expected[rule])
        fp = len(flagged[rule] - expected[rule])
        fn = len(expected[rule] - flagged[rule])
        rows.append({"rule_id": rule, "expected": len(expected[rule]),
                     "flagged": len(flagged[rule]), "tp": tp, "fp": fp, "fn": fn,
                     "precision": round(tp / (tp + fp), 4) if tp + fp else 1.0,
                     "recall": round(tp / (tp + fn), 4) if tp + fn else 1.0})
    return rows


==================== docs/data_dictionary.md ====================

# Data Dictionary

All data is synthetic, including identities and bank and tax details. IDs are immutable and zero-padded. Timestamps are UTC ISO 8601, dates are YYYY-MM-DD, and money is in INR with 2 decimals.

## Employee
Master table. One row per employee. Pay group changes and other history are kept as events in later tables, so this row holds the current state only.

| Field | Type | Required | Allowed / format | Unique | Validation |
|---|---|---|---|---|---|
| employee_id | string | yes | EMP-000001 | PK | matches ^EMP-\d{6}$, immutable |
| first_name | string | yes | synthetic name list | no | non-empty |
| last_name | string | yes | synthetic name list | no | non-empty |
| email | string | yes | synthetic, @example.com | yes | valid format, lowercase |
| role | enum | yes | employee, manager, hr_partner | no | one of the allowed values |
| manager_id | string | optional | existing EMP- id | no | FK to Employee (self); null for top-level managers; cannot be the employee's own id |
| pay_group_id | string | yes | existing PG- id | no | FK to PayGroup |
| pay_type | enum | yes | hourly, salaried | no | one of the allowed values |
| hourly_rate | decimal(10,2) | optional | INR per hour | no | required and greater than 0 when pay_type is hourly, null otherwise |
| monthly_salary | decimal(12,2) | optional | INR per month | no | required and greater than 0 when pay_type is salaried, null otherwise |
| hire_date | date | yes | YYYY-MM-DD | no | before the first pay period start for staff hired before the data begins |
| termination_date | date | optional | YYYY-MM-DD | no | null for active staff; after hire_date when set |
| status | enum | yes | active, terminated | no | terminated if and only if termination_date is set |
| bank_account_number | string | optional | synthetic, 12 digits | no | required before a payroll export; a missing value is a planted validation case |
| bank_ifsc | string | optional | synthetic, 4 letters + 0 + 6 digits | no | required together with bank_account_number |
| tax_id | string | optional | synthetic, TIN- + 8 digits | yes | required before a payroll export; a missing value is a planted validation case |

Notes:
- Bank and tax fields are optional on purpose. The workflow checks for them before export, so a baseline row can lack them only when a ground-truth label says so.
- manager_id gives the approval chain, because an employee's time and leave are approved by their manager and payroll inputs by an hr_partner.
- Bank accounts and tax IDs use fictional formats, so no real person or account can match them.


## PayGroup
Master table. A pay group defines the pay rules shared by its employees. Employee.pay_group_id points here (1:N from PayGroup to Employee).

| Field | Type | Required | Allowed / format | Unique | Validation |
|---|---|---|---|---|---|
| pay_group_id | string | yes | PG-001 | PK | matches ^PG-\d{3}$, immutable |
| name | string | yes | e.g. Corporate, Operations, Contract | yes | non-empty |
| pay_frequency | enum | yes | monthly | no | one of the allowed values |
| standard_hours_per_day | decimal(4,2) | yes | hours | no | greater than 0 and at most 12 |
| max_hours_per_period | decimal(6,2) | yes | hours per pay period | no | greater than 0; hours above this are flagged as exceeding the period maximum |
| overtime_multiplier | decimal(3,2) | yes | multiple of the hourly rate | no | at least 1.00 |
| payday_day | integer | yes | 1 to 28 | no | day of the month after the period ends on which pay is released |
| effective_from | date | yes | YYYY-MM-DD | no | on or before the first pay period start |
| status | enum | yes | active, inactive | no | one of the allowed values |

Notes:
- All three pay groups are active for the whole data period. Pay rule changes would be new rows with a later effective_from, never edits to an old row.
- Whether an employee is paid hourly or a fixed salary is on Employee (pay_type), not here, so each group can mix both.



## PayPeriod
Master table. One shared monthly calendar for all pay groups (3 rows: 2026-01, 2026-02, 2026-03). Each period is a parent of the time, leave, allowance, deduction, payroll input and export records for that month.

| Field | Type | Required | Allowed / format | Unique | Validation |
|---|---|---|---|---|---|
| pay_period_id | string | yes | PP-2026-01 | PK | matches ^PP-\d{4}-\d{2}$, immutable |
| start_date | date | yes | YYYY-MM-DD | yes | first day of the month |
| end_date | date | yes | YYYY-MM-DD | yes | last day of the month; after start_date |
| working_days | integer | yes | days | no | Monday to Friday days in the period; greater than 0 |
| submission_deadline | date | yes | YYYY-MM-DD | no | after end_date; time and leave entered after this are late |
| status | enum | yes | open, validating, approved, exported | no | one of the allowed values; moves forward only |

Lifecycle: open -> validating -> approved -> exported. A period can only be exported after it is approved. Each move is recorded as an Approval or PayrollExport row, so the history is never overwritten and status only shows the current state.

Notes:
- Periods must not overlap and must follow each other without gaps.
- Overtime and hour limits are per pay group (max_hours_per_period on PayGroup), not per period.


## TimeEntry
Transaction table. One row per hours submission by an hourly employee. Rows are never edited: a correction or a duplicate submission is a new row that points to the original. History is preserved in full.

| Field | Type | Required | Allowed / format | Unique | Validation |
|---|---|---|---|---|---|
| time_entry_id | string | yes | TIME-EMP-000001-2026-01-05, with -C1 (correction) or -D1 (duplicate) suffix | PK | matches ^TIME-EMP-\d{6}-\d{4}-\d{2}-\d{2}(-[CD]\d)?$; the prefix must equal the employee_id and work_date; immutable |
| employee_id | string | yes | existing EMP- id | no | FK to Employee; the employee's pay_type must be hourly |
| pay_period_id | string | yes | existing PP- id | no | FK to PayPeriod; work_date must fall inside the period |
| work_date | date | yes | YYYY-MM-DD | no | on or after hire_date and not after termination_date |
| hours_worked | decimal(4,2) | yes | hours | no | greater than 0 and at most 24 |
| entry_type | enum | yes | original, correction, duplicate_submission | no | one of the allowed values |
| related_entry_id | string | optional | existing TIME- id | no | FK to TimeEntry (self); required for correction and duplicate_submission and null for original; must point to an original entry of the same employee and work_date |
| submitted_at | timestamp | yes | UTC | no | on or after work_date; a submission after the period's submission_deadline is late |
| approval_status | enum | yes | approved, pending, rejected | no | one of the allowed values; set when the row is created |
| approved_by | string | optional | existing EMP- id | no | FK to Employee; required when approved or rejected; must be the employee's manager_id |
| approved_at | timestamp | optional | UTC | no | required when approved_by is set; on or after submitted_at |

Uniqueness rules beyond the primary key:
- At most one original entry per (employee_id, work_date).
- At most one correction per original entry.

Effective hours for an employee on a day: the correction if it is approved, otherwise the original if it is approved. A duplicate_submission never counts toward hours. Duplicates are expected in the baseline data and are caught by the duplicate-hours validation rule. Overtime is calculated during aggregation, never stored here.


## LeaveRecord
Transaction table. One row per leave request for a salaried employee, always inside a single pay period. Rows are never edited: a change is a new correction row that points to the original.

| Field | Type | Required | Allowed / format | Unique | Validation |
|---|---|---|---|---|---|
| leave_id | string | yes | LV-000001 | PK | matches ^LV-\d{6}$, immutable |
| employee_id | string | yes | existing EMP- id | no | FK to Employee; the employee's pay_type must be salaried |
| pay_period_id | string | yes | existing PP- id | no | FK to PayPeriod; start_date and end_date must both fall inside the period |
| leave_type | enum | yes | annual, sick, unpaid | no | one of the allowed values |
| start_date | date | yes | YYYY-MM-DD | no | on or after hire_date and not after termination_date |
| end_date | date | yes | YYYY-MM-DD | no | on or after start_date |
| days | integer | yes | working days (Monday to Friday) | no | equals the number of working days from start_date to end_date inclusive; at least 1 |
| entry_type | enum | yes | original, correction | no | one of the allowed values |
| related_leave_id | string | optional | existing LV- id | no | FK to LeaveRecord (self); required for correction and null for original; must point to an original record of the same employee |
| submitted_at | timestamp | yes | UTC | no | a late submission is one after the period's submission_deadline |
| approval_status | enum | yes | approved, pending, rejected | no | one of the allowed values; set when the row is created |
| approved_by | string | optional | existing EMP- id | no | FK to Employee; required when approved or rejected; must be the employee's manager_id |
| approved_at | timestamp | optional | UTC | no | required when approved_by is set; on or after submitted_at |

Rules:
- Approved leave records of one employee must not overlap in dates, apart from a correction replacing its original.
- Effective leave is the correction if approved, otherwise the original if approved. Pending and rejected rows never count.
- Only effective unpaid leave produces a deduction. Each effective unpaid leave record has exactly one unpaid_leave Deduction, linked by leave_id. A second one is the "leave deduction applied twice" anomaly.
- Paid leave (annual, sick) reduces no pay.


## Allowance
Transaction table. One row per allowance granted to an employee for a pay period. Rows are never edited: a change is a new correction row that points to the original.

| Field | Type | Required | Allowed / format | Unique | Validation |
|---|---|---|---|---|---|
| allowance_id | string | yes | ALW-000001 | PK | matches ^ALW-\d{6}$, immutable |
| employee_id | string | yes | existing EMP- id | no | FK to Employee; must be eligible for the allowance type (see rules) |
| pay_period_id | string | yes | existing PP- id | no | FK to PayPeriod |
| allowance_type | enum | yes | transport, meal, remote_work, shift | no | one of the allowed values |
| units | integer | yes | count (days or 1) | no | at least 1; for meal and shift equals the days with effective time entries in the period |
| unit_amount | decimal(10,2) | yes | INR per unit | no | equals the configured rate for the allowance type |
| amount | decimal(12,2) | yes | INR | no | equals units x unit_amount |
| entry_type | enum | yes | original, correction | no | one of the allowed values |
| related_allowance_id | string | optional | existing ALW- id | no | FK to Allowance (self); required for correction and null for original; must point to an original of the same employee, period and type |
| created_at | timestamp | yes | UTC | no | on or after the pay period start |
| approval_status | enum | yes | approved, pending, rejected | no | one of the allowed values; set when the row is created |
| approved_by | string | optional | existing EMP- id | no | FK to Employee; required when approved or rejected; must be the employee's manager_id |
| approved_at | timestamp | optional | UTC | no | required when approved_by is set; on or after created_at |

Eligibility rules (defined in config.py and decisions.md, checked by a validation rule):
- transport: all active employees, 1 unit per period, INR 2,000.
- meal: hourly employees only, 1 unit per day worked, INR 100.
- remote_work: salaried employees in the Corporate pay group, 1 unit per period, INR 1,500.
- shift: hourly employees in the Operations pay group, 1 unit per day worked, INR 250.

Uniqueness rules beyond the primary key:
- At most one original per (employee_id, pay_period_id, allowance_type). A second original is the "same allowance duplicated" anomaly.
- At most one correction per original.

Effective allowance: the correction if approved, otherwise the original if approved. Pending and rejected rows never count toward payroll.

For the group heads, who have no manager, approved_by is an hr_partner.


## PayrollInput
Derived table. One row per employee per pay period, aggregated from the effective approved source records. No value is generated independently, and every amount can be recomputed from the source tables.

| Field | Type | Required | Allowed / format | Unique | Validation |
|---|---|---|---|---|---|
| payroll_input_id | string | yes | PIN-EMP-000001-2026-01 | PK | matches ^PIN-EMP-\d{6}-\d{4}-\d{2}$; must equal PIN-<employee_id>-<year>-<month of the period>; immutable |
| employee_id | string | yes | existing EMP- id | no | FK to Employee; unique together with pay_period_id |
| pay_period_id | string | yes | existing PP- id | no | FK to PayPeriod |
| pay_group_id | string | yes | existing PG- id | no | FK to PayGroup; the employee's pay group for this period |
| pay_type | enum | yes | hourly, salaried | no | equals the employee's pay_type |
| regular_hours | decimal(6,2) | yes | hours | no | sum over effective approved time entries of min(hours, standard_hours_per_day) per day; 0 for salaried |
| overtime_hours | decimal(6,2) | yes | hours | no | sum of hours above standard_hours_per_day per day; 0 for salaried |
| total_hours | decimal(6,2) | yes | hours | no | equals regular_hours + overtime_hours; flagged when above the pay group's max_hours_per_period |
| base_pay | decimal(12,2) | yes | INR | no | hourly: regular_hours x hourly_rate; salaried: monthly_salary |
| overtime_pay | decimal(12,2) | yes | INR | no | overtime_hours x hourly_rate x overtime_multiplier; 0 for salaried |
| allowances_total | decimal(12,2) | yes | INR | no | sum of effective approved Allowance amounts |
| deductions_total | decimal(12,2) | yes | INR | no | sum of effective approved Deduction amounts |
| gross_pay | decimal(12,2) | yes | INR | no | equals base_pay + overtime_pay + allowances_total |
| net_pay | decimal(12,2) | yes | INR | no | equals gross_pay - deductions_total; must not be negative |
| unpaid_leave_days | decimal(5,2) | yes | days | no | sum of days of effective approved unpaid leave; for information, not subtracted again |
| time_entry_count | integer | yes | count | no | number of effective approved time entries used |
| allowance_count | integer | yes | count | no | number of effective approved allowances used |
| deduction_count | integer | yes | count | no | number of effective approved deductions used |
| status | enum | yes | calculated, flagged, approved, exported | no | one of the allowed values; moves forward only, except that flagged returns to calculated when its exceptions are resolved |
| calculated_at | timestamp | yes | UTC | no | on or after the pay period's end_date |

Lifecycle: calculated -> flagged (one or more open Exceptions) -> calculated (resolved) -> approved (HR partner Approval) -> exported (PayrollExport). A row can only become approved when it has no open exceptions and every source record it used is approved. Each transition is stored as an Approval or PayrollExport row.

Reconciliation rules (checked by validate.py):
- Every amount and count equals a fresh recomputation from the source tables.
- For each pay period, the sum of net_pay over exported rows equals the total of its PayrollExport.
- Source records that are pending, rejected, or superseded by a correction never contribute to any total.



## ValidationRule
Master table. The catalogue of validation checks run before payroll export. One row per rule version. Rows are never edited: a change to a rule or its threshold is a new row with a later effective_from, and the old row is set to retired by that new row's existence (the status of the old row is not rewritten; the current rule is the one with the latest effective_from per rule_code).

| Field | Type | Required | Allowed / format | Unique | Validation |
|---|---|---|---|---|---|
| rule_id | string | yes | VR-001 | PK | matches ^VR-\d{3}$, immutable |
| rule_code | string | yes | UPPER_SNAKE_CASE, e.g. HOURS_EXCEED_PERIOD_MAX | no | unique together with effective_from |
| name | string | yes | short title | no | non-empty |
| description | string | yes | what the rule checks, in one sentence | no | non-empty |
| applies_to | enum | yes | employee, time_entry, leave_record, allowance, deduction, payroll_input, payroll_export | no | one of the allowed values |
| severity | enum | yes | blocking, warning | no | one of the allowed values |
| threshold_value | decimal(10,2) | optional | number | no | required for rules that have a limit (see catalogue), null otherwise |
| threshold_unit | string | optional | e.g. hours, INR | no | required when threshold_value is set |
| effective_from | date | yes | YYYY-MM-DD | no | on or before the first pay period start |

Severity effect:
- blocking: the Exception keeps the PayrollInput in flagged status, so it cannot be approved or exported until the exception is resolved.
- warning: the Exception is recorded and reviewed, but does not stop approval or export.

Initial catalogue (all effective from the first pay period start):

| rule_id | rule_code | applies_to | severity | Threshold | What it checks |
|---|---|---|---|---|---|
| VR-001 | MISSING_BANK_DETAILS | employee | blocking | none | bank_account_number or bank_ifsc is missing |
| VR-002 | MISSING_TAX_ID | employee | blocking | none | tax_id is missing |
| VR-003 | DUPLICATE_HOURS | time_entry | blocking | none | a duplicate_submission exists for an employee and day that already has an original or correction |
| VR-004 | HOURS_EXCEED_PERIOD_MAX | payroll_input | blocking | pay group's max_hours_per_period | total_hours is above the pay group maximum |
| VR-005 | DAILY_HOURS_OUTLIER | time_entry | warning | 14 hours | one day's hours are above the threshold |
| VR-006 | DUPLICATE_ALLOWANCE | allowance | blocking | none | more than one original allowance of the same type for an employee and period |
| VR-007 | INELIGIBLE_ALLOWANCE | allowance | blocking | none | the employee does not meet the eligibility rule for the allowance type |
| VR-008 | LEAVE_DEDUCTION_DUPLICATED | deduction | blocking | none | more than one unpaid_leave deduction for the same leave_id |
| VR-009 | UNAPPROVED_SOURCE_USED | payroll_input | blocking | none | a payroll input or export includes a source record that is not approved |
| VR-010 | TOTALS_DO_NOT_RECONCILE | payroll_input | blocking | 0.01 INR | a payroll input amount differs from the recomputed sum of its source records by more than the threshold |
| VR-011 | NEGATIVE_NET_PAY | payroll_input | blocking | 0 INR | net_pay is below zero |
| VR-012 | LATE_SUBMISSION | time_entry | warning | none | a time entry or leave record was submitted after the period's submission_deadline |

Notes:
- The five required edge cases are detected by VR-004, VR-006, VR-009, VR-008 and VR-010.
- VR-009 and VR-010 are re-run at export time as a second gate, so an export can never contain an unapproved source or a total that does not reconcile.


## Exception
Event table. One row per finding raised by a validation rule on a source record. The row holds the current status only; each change of status is recorded as an Approval row, so history is never overwritten.

| Field | Type | Required | Allowed / format | Unique | Validation |
|---|---|---|---|---|---|
| exception_id | string | yes | EXC-000001 | PK | matches ^EXC-\d{6}$, immutable |
| rule_id | string | yes | existing VR- id | no | FK to ValidationRule; the rule version in force at detected_at |
| employee_id | string | yes | existing EMP- id | no | FK to Employee |
| pay_period_id | string | yes | existing PP- id | no | FK to PayPeriod |
| payroll_input_id | string | optional | existing PIN- id | no | FK to PayrollInput; required when the exception blocks a payroll input, null for employee-level findings that exist before any input is built |
| source_table | enum | yes | employees, time_entries, leave_records, allowances, deductions, payroll_inputs, payroll_exports | no | one of the allowed values; must match the rule's applies_to |
| source_record_id | string | yes | id of the flagged record | no | must exist in source_table; checked by validate.py |
| detected_at | timestamp | yes | UTC | no | on or after the pay period's end_date |
| detail | string | yes | one sentence with the values that triggered the rule | no | non-empty |
| status | enum | yes | open, resolved, waived | no | one of the allowed values; see lifecycle |

Lifecycle: open -> resolved, or open -> waived. Each move creates an Approval row of type exception_resolved or exception_waived, so the status is the current state and the history is in Approval.
- resolved: the underlying data was fixed (a correction or rejection was recorded) and the rule passes on re-run.
- waived: allowed only when the rule's severity is warning, and only by an hr_partner.
- A blocking exception can never be waived.

Uniqueness rules beyond the primary key:
- At most one open Exception per (rule_id, source_record_id). A re-run of the rules must not create a second open copy of the same finding.

Effect on payroll:
- A PayrollInput with at least one open blocking Exception has status flagged and cannot be approved or exported.
- When the last open blocking Exception is resolved, the PayrollInput returns to calculated and is recomputed.


## Approval
Event table, append-only. One row per payroll-level decision. Source-record approvals (time, leave, allowance, deduction) are stored on those rows and are not repeated here. Rows are never edited or deleted.

| Field | Type | Required | Allowed / format | Unique | Validation |
|---|---|---|---|---|---|
| approval_id | string | yes | APR-000001 | PK | matches ^APR-\d{6}$, immutable |
| approval_type | enum | yes | period_validation_started, payroll_input_approved, exception_resolved, exception_waived, period_approved | no | one of the allowed values |
| pay_period_id | string | yes | existing PP- id | no | FK to PayPeriod |
| payroll_input_id | string | optional | existing PIN- id | no | FK to PayrollInput; required for payroll_input_approved, null otherwise |
| exception_id | string | optional | existing EXC- id | no | FK to Exception; required for exception_resolved and exception_waived, null otherwise |
| decided_by | string | yes | existing EMP- id | no | FK to Employee; must have the hr_partner role |
| decided_at | timestamp | yes | UTC | no | on or after the pay period's end_date; later than the exception's detected_at when exception_id is set |
| comment | string | optional | free text | no | required for exception_waived (the reason) |

Rules:
- payroll_input_approved: the input must have status calculated, no open blocking exceptions, and every source record it used must be approved. decided_by must not be the input's own employee_id.
- exception_waived: allowed only when the exception's rule severity is warning. A blocking exception can never be waived.
- period_approved: allowed only when no PayrollInput of the period is still in status calculated. Every input is approved, exported, or flagged and held back from export.
- period_validation_started and period_approved move PayPeriod.status from open to validating and from validating to approved.
- Every change of status on PayPeriod, PayrollInput and Exception has exactly one Approval row (or a PayrollExport row for exported), so history is never overwritten.


## PayrollExport
Event table. One row per export batch for a pay period. Only approved payroll inputs with no open blocking exceptions are included. A period can have several batches, because inputs held back by an exception can be exported later; each payroll input appears in at most one batch. Rows are never edited.

| Field | Type | Required | Allowed / format | Unique | Validation |
|---|---|---|---|---|---|
| export_id | string | yes | PEX-2026-000001 | PK | matches ^PEX-\d{4}-\d{6}$, immutable |
| pay_period_id | string | yes | existing PP- id | no | FK to PayPeriod; the period's status must be approved or exported at exported_at |
| exported_by | string | yes | existing EMP- id | no | FK to Employee; must have the hr_partner role |
| exported_at | timestamp | yes | UTC | no | on or after the latest Approval of the payroll inputs it contains |
| record_count | integer | yes | count | no | equals the number of PayrollExportLine rows for this export |
| total_gross | decimal(14,2) | yes | INR | no | equals the sum of gross_pay of its lines' payroll inputs |
| total_deductions | decimal(14,2) | yes | INR | no | equals the sum of deductions_total of its lines' payroll inputs |
| total_net | decimal(14,2) | yes | INR | no | equals total_gross - total_deductions and the sum of its lines' net_pay |
| status | enum | yes | completed | no | one of the allowed values |

Rules:
- The export gate runs rules VR-009 and VR-010 again at export time, so no unapproved source record and no non-reconciling total can leave the system.
- Exporting a batch moves each included PayrollInput to exported. When the first batch of a period is created, the period's status moves from approved to exported and stays there.
- The anomaly "payroll totals fail to reconcile" is a batch whose totals differ from the sum of its lines or inputs.

# Decisions and Assumptions

Each item fills a gap in the problem statement or fixes a rule. Code, validation and metrics must follow this file. The problem statement is paraphrased here, not copied.

## 1. Architecture (three tiers)
- Database: SQLite (data/payroll.db) holds all entities. CSV, JSON and SQL exports are written beside it.
- Backend: Python (standard library generators and rules) plus a FastAPI app that exposes the data, validation, approvals and export gate. CLI commands: generate, validate, reproduce.
- Frontend: React with Vite. Four screens: Dashboard, Exceptions, Approvals, Exports. It only reads and acts through the API.
- Generic scaffolding (ID counters, exporter and manifest, validator framework) is reused from my own earlier project, cart-recovery-datagen. All payroll domain logic is new.

## 2. Reproducibility
- Fixed seed 20261022 through a single random.Random instance. No wall-clock time is used during generation.
- Scale is parameterized (starter and full share one generator).
- Rows are exported sorted by primary key, with a manifest of row counts and SHA-256 checksums. reproduce.py generates twice and compares.
- Money uses Decimal with ROUND_HALF_UP to 2 decimals, so totals reconcile exactly.

## 3. Data period and calendar
- Three monthly pay periods: PP-2026-01, PP-2026-02, PP-2026-03 (January to March 2026), shared by all pay groups.
- working_days counts Monday to Friday. There is no holiday calendar.
- submission_deadline is 5 days after the period end. Anything submitted later is late.

## 4. Volumes (full dataset)
- 3,000 employees in 3 pay groups: Corporate, Operations, Contract (PG-001 to PG-003).
- About 38% hourly and 62% salaried. Only hourly staff have time entries, only salaried staff have leave.
- The statement asks for about 100,000 time, allowance and deduction records over three months. Volume follows from the rules (one time entry per hourly employee per working day, plus allowances and deductions) and is tuned by the hourly share to land within 10% of 100,000. The exact counts are reported in the validation summary.
- Starter dataset: 300 employees, same rules.

## 5. Employees
- All identities, emails (@example.com), bank accounts (12 digits, IFSC-style) and tax IDs (TIN- plus 8 digits) are fictional.
- About 2% of employees are hired or terminated during the period, so some periods are partial.
- manager_id forms the approval chain for source records. HR partners (about 1% of staff) approve payroll inputs and exceptions.
- Managers and HR partners are salaried. HR partners are all in the Corporate pay group. The first manager of each pay group is its head and has no manager; heads take no leave, because it would have no approver. Every other employee has a manager from their own pay group.
- Missing bank and tax fields are planted with fixed counts (at least 2 each), not by chance, and labelled in the ground truth with the rule expected to catch them (VR-001, VR-002).

## 6. Source records and approval
- Time, leave, allowance and deduction rows carry their own approval fields, set when the row is created. Payroll-level decisions (payroll inputs, exceptions, period) live in Approval.
- About 93% of source rows are approved, 5% pending and 2% rejected. Pending and rejected rows never count toward payroll.
- Corrections and duplicate submissions are new rows pointing to the original. Effective value = the approved correction if there is one, otherwise the approved original. A duplicate_submission never counts toward hours.
- Baseline: about 2% of time entries get a correction and about 1% get a duplicate submission.
- Hours are quarter-hour values. About 4% of working days have no entry (absence). A submission is late when its date is after the period's submission_deadline. Corrections are made only for approved, on-time, non-outlier originals; duplicate submissions copy the original's hours and never count. Rows are conservatively capped so no baseline employee exceeds the pay group maximum; exceeding it happens only through the labelled injection.
- Leave: only salaried employees who have a manager (group heads take none). At most one unpaid and one paid block per employee per period, placed with at least 2 free working days between blocks so records never overlap and a correction of plus or minus one day never collides. Leave is never submitted late. Effective leave is the approved correction, otherwise the approved original.

- Allowances and deductions have no corrections in the baseline, because their units and amounts are fully derived. Allowances are approved by the employee's manager; the three group heads have no manager, so an HR partner approves theirs. Deductions are approved by an HR partner. Provident fund, professional tax and unpaid-leave deductions are always approved (they are automatic); only loan_recovery follows the 93/5/2 approval split.
- An allowance or deduction is created on the first or second day after the period ends. An employee is active in a period if hired before it ends and not terminated before it starts.

## 7. Pay calculation (all derived)
- Overtime is hours above the pay group's standard_hours_per_day on each day, paid at hourly_rate x overtime_multiplier.
- Hourly base_pay = regular_hours x hourly_rate. Salaried base_pay = monthly_salary.
- Unpaid leave (salaried only) reduces pay only through its Deduction row: units = leave days, unit_amount = monthly_salary / working_days.
- Allowance rules: transport (all active, INR 2,000 per period), meal (hourly, INR 100 per day worked), remote_work (salaried in Corporate, INR 1,500 per period), shift (hourly in Operations, INR 250 per day worked).
- Deduction rules: provident_fund (salaried, 12% of monthly_salary), professional_tax (all active, INR 200), loan_recovery (about 8% of employees, a fixed installment), unpaid_leave (one per effective unpaid leave record).
- gross_pay = base_pay + overtime_pay + allowances_total. net_pay = gross_pay - deductions_total and must not be negative.

## 8. Validation rules (stored in ValidationRule)
- 12 rules, VR-001 to VR-012 (see the data dictionary for the catalogue). Blocking rules keep the payroll input in flagged status. Warning rules are recorded and reviewed.
- Thresholds: daily hours outlier 14 hours, reconciliation tolerance 0.01 INR, net pay below 0.
- Rules run after aggregation and again at export (VR-009 and VR-010 as a second gate).

## 9. Exceptions, approvals and export
- Each finding becomes one Exception row. About 85% of blocking exceptions get resolved through a correction or rejection, and their payroll inputs are recalculated. The rest stay open, so their inputs are held back from export.
- Warning exceptions are waived by an HR partner or left open. They do not block export.
- An HR partner approves clean payroll inputs, never their own. A period is approved when no input remains in calculated status.
- Export happens in batches. Only approved inputs with no open blocking exception are exported, each in at most one batch.

## 10. Outcomes and labels
- Every outcome (flags, approvals, resolutions, export inclusion) comes from the rules above and fixed probabilities that depend on the record, never from independent random assignment.
- Baseline-planted validation cases (missing bank or tax field, duplicate hours, late submission, daily outliers) are generated first, flagged by the rules, and labelled in the ground truth.

## 11. Labelled anomalies (injected after the valid baseline)
1. hours_exceed_period_max: an employee's total hours go above the pay group maximum.
2. duplicate_allowance: a second original allowance of the same type for one employee and period.
3. unapproved_time_exported: an export line whose payroll input used an unapproved time entry.
4. leave_deduction_twice: a second unpaid_leave deduction for the same leave record.
5. totals_not_reconciled: an export batch (or payroll input) whose total differs from the sum of its parts.
- Counts: 3 per type in the starter set and 40 per type in the full set. Each injection touches only its target rows and is recorded in ground_truth with the rule expected to catch it.
- ground_truth is a separate output and is never read by any generator, rule or feature step. validate.py runs every check first and compares with the labels afterwards.

## 12. ID conventions
- Given: EMP-000001, PP-2026-01, TIME-<employee-id>-<date>, PIN-<employee-id>-2026-01, PEX-2026-000001.
- Added by us: PG-001, LV-, ALW-, DED-, VR-, EXC-, APR-, PXL-.
- TimeEntry corrections and duplicates add the suffix -C1 or -D1 to the original ID.
- Counters are zero-padded and IDs are immutable.

## 13. Deviations and additions
- Added the supporting table PayrollExportLine so every exported amount is traceable.
- No bridge table: Exception links rules to the records they flag.
- ValidationRule changes are new rows with a later effective_from. Old rows are never edited.

## 14. Provenance and lawful use
- No external or real data. Everything is generated. Names come from a short built-in list.
- Prompts to AI tools are kept in ai-prompts/prompt-history.md.
